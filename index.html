// 向 GAS 抓取資料並繪製表格 async function loadSheetData() { try { const response = await fetch(GAS_API_URL); const result = await response.json(); // 取得 GAS 傳回的分頁名稱與資料 const sheetName = result.sheetName || "班表"; originalData = result.data || []; // 更新網頁標題 document.querySelector("h2").innerText = `📅 雲端班表 (${sheetName})`; const tbody = document.getElementById("table-body"); tbody.innerHTML = ""; originalData.forEach((row, index) => { const actualRowIndex = index + 2; // H2:I62 的第一列對應第 2 行 const canEdit = isRowEditable(actualRowIndex); const tr = document.createElement("tr"); const tdRow = document.createElement("td"); tdRow.innerText = actualRowIndex; tr.appendChild(tdRow); ['H', 'I'].forEach((col, colIndex) => { const td = document.createElement("td"); const input = document.createElement("input"); input.type = "text"; input.value = row[colIndex] || ""; input.id = `${col}${actualRowIndex}`; input.className = "cell-input" + (canEdit ? " editable" : ""); if (!canEdit) { input.disabled = true; } td.appendChild(input); tr.appendChild(td); }); tbody.appendChild(tr); }); document.getElementById("save-btn").disabled = false; } catch (err) { alert("載入班表失敗：" + err.message); } }
